// --- API Helper ---
const API_KEY = ""; // Dynamic key or default empty string

async function callGemini({ prompt, imageBase64, mimeType, systemInstruction }) {
  const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=${API_KEY}`;
  
  const parts = [];
  if (prompt) {
    parts.push({ text: prompt });
  }
  
  if (imageBase64 && mimeType) {
    parts.push({
      inlineData: {
        mimeType: mimeType,
        data: imageBase64
      }
    });
  }

  const payload = {
    contents: [{ role: "user", parts }],
  };

  if (systemInstruction) {
    payload.systemInstruction = {
      parts: [{ text: systemInstruction }]
    };
  }

  // Exponential backoff logic for retries
  let delay = 1000;
  for (let attempt = 0; attempt < 3; attempt++) {
    try {
      const response = await fetch(apiUrl, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload)
      });

      if (!response.ok) {
        throw new Error(`HTTP error! status: ${response.status}`);
      }

      const data = await response.json();
      const textResult = data.candidates?.[0]?.content?.parts?.[0]?.text;
      
      if (!textResult) {
        throw new Error("Geçerli bir yanıt alınamadı.");
      }

      return textResult;
    } catch (err) {
      if (attempt === 2) throw err;
      await new Promise(res => setTimeout(res, delay));
      delay *= 2;
    }
  }
}

// --- Simple Custom Markdown Renderer Component ---
function SimpleMarkdown({ content }) {
  if (!content) return null;

  // Basic formatting parsing for bold, code, lists, and line breaks
  const lines = content.split('\n');

  return (
    <div className="space-y-2 text-slate-700 leading-relaxed text-sm md:text-base">
      {lines.map((line, idx) => {
        let trimmed = line.trim();

        if (trimmed.startsWith('### ')) {
          return <h3 key={idx} className="text-lg font-bold text-slate-800 mt-4 mb-2">{trimmed.replace('### ', '')}</h3>;
        }
        if (trimmed.startsWith('## ')) {
          return <h2 key={idx} className="text-xl font-bold text-slate-800 mt-5 mb-2 border-b border-slate-100 pb-1">{trimmed.replace('## ', '')}</h2>;
        }
        if (trimmed.startsWith('# ')) {
          return <h1 key={idx} className="text-2xl font-extrabold text-slate-900 mt-6 mb-3">{trimmed.replace('# ', '')}</h1>;
        }
        if (trimmed.startsWith('- ') || trimmed.startsWith('* ')) {
          const listText = trimmed.substring(2);
          return (
            <div key={idx} className="flex items-start gap-2 my-1 pl-2">
              <span className="inline-block w-1.5 h-1.5 rounded-full bg-indigo-500 mt-2.5 flex-shrink-0"></span>
              <span dangerouslySetInnerHTML={{ __html: formatInlineMarkdown(listText) }} />
            </div>
          );
        }
        if (/^\d+\.\s/.test(trimmed)) {
          const match = trimmed.match(/^(\d+)\.\s(.*)/);
          return (
            <div key={idx} className="flex items-start gap-2.5 my-1.5 pl-1">
              <span className="font-semibold text-indigo-600 bg-indigo-50 text-xs px-2 py-0.5 rounded-md mt-0.5">{match[1]}</span>
              <span dangerouslySetInnerHTML={{ __html: formatInlineMarkdown(match[2]) }} />
            </div>
          );
        }

        if (trimmed === '') return <div key={idx} className="h-2"></div>;

        return (
          <p key={idx} dangerouslySetInnerHTML={{ __html: formatInlineMarkdown(line) }} />
        );
      })}
    </div>
  );
}

function formatInlineMarkdown(text) {
  return text
    .replace(/\*\*(.*?)\*\*/g, '<strong class="font-semibold text-slate-900">$1</strong>')
    .replace(/\*(.*?)\*/g, '<em class="italic text-slate-800">$1</em>')
    .replace(/`(.*?)`/g, '<code class="bg-slate-100 text-indigo-600 font-mono text-xs px-1.5 py-0.5 rounded border border-slate-200">$1</code>');
}

                  </button>
                </div>
              </div>

              {/* Result Area */}
              {imageResult && (
                <div className="bg-white rounded-2xl p-6 border border-slate-200/80 shadow-sm space-y-3">
                  <div className="flex items-center justify-between border-b border-slate-100 pb-3">
                    <span className="text-xs font-semibold text-violet-600 uppercase tracking-wider flex items-center gap-1.5">
                      <Check className="w-4 h-4 text-violet-500" /> Görsel Analiz Çözümü
                    </span>
                    <button
                      onClick={() => handleCopy(imageResult)}
                      className="text-xs text-slate-400 hover:text-slate-600 flex items-center gap-1 bg-slate-50 hover:bg-slate-100 px-2.5 py-1 rounded-lg transition-all"
                    >
                      {copied ? <Check className="w-3.5 h-3.5 text-emerald-500" /> : <Copy className="w-3.5 h-3.5" />}
                      {copied ? "Kopyalandı" : "Kopyala"}
                    </button>
                  </div>
                  <SimpleMarkdown content={imageResult} />
                </div>
              )}
            </div>
          )}

          {/* TAB 3: Konu Anlatımı */}
          {activeTab === 'topic' && (
            <div className="space-y-6 animate-fadeIn">
              <header className="space-y-1">
                <div className="inline-flex items-center gap-1.5 px-2.5 py-1 rounded-full bg-emerald-50 text-emerald-600 text-xs font-semibold">
                  <BookOpen className="w-3.5 h-3.5" /> Anlaşılır Dersler
                </div>
                <h2 className="text-2xl font-bold text-slate-900">Kolay Konu Anlatımı</h2>
                <p className="text-slate-500 text-sm">Anlamakta zorlandığınız konuları günlük hayat benzetmeleri ve basit adımlarla öğrenin.</p>
              </header>

              <div className="bg-white rounded-2xl p-5 border border-slate-200/80 shadow-sm space-y-4">
                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
                  <div>
                    <label className="block text-xs font-semibold text-slate-600 mb-1.5">Konu Adı</label>
                    <input
                      type="text"
                      value={topicName}
                      onChange={(e) => setTopicName(e.target.value)}
                      placeholder="Örn: Newton'un Hareket Yasaları, Mitoz Bölünme..."
                      className="w-full bg-slate-50 border border-slate-200 rounded-xl px-3 py-2 text-sm focus:outline-none focus:ring-2 focus:ring-indigo-500/20 focus:border-indigo-500"
                    />
                  </div>

                  <div>
                    <label className="block text-xs font-semibold text-slate-600 mb-1.5">Sınıf / Seviye</label>
                    <select 
                      value={gradeLevel} 
                      onChange={(e) => setGradeLevel(e.target.value)}
                      className="w-full bg-slate-50 border border-slate-200 rounded-xl px-3 py-2 text-sm focus:outline-none focus:ring-2 focus:ring-indigo-500/20 focus:border-indigo-500"
                    >
                      <option value="İlkokul">İlkokul</option>
                      <option value="Ortaokul">Ortaokul</option>
                      <option value="Lise">Lise</option>
                      <option value="Üniversite">Üniversite</option>
                    </select>
                  </div>
                </div>

                <div>
                  <label className="block text-xs font-semibold text-slate-600 mb-1.5">Özellikle Karıştırılan Yer (Opsiyonel)</label>
                  <input
                    type="text"
                    value={topicDetail}
                    onChange={(e) => setTopicDetail(e.target.value)}
                    placeholder="Örn: Anafaz ve Telofaz evrelerinin farkını anlamıyorum..."
                    className="w-full bg-slate-50 border border-slate-200 rounded-xl px-3 py-2 text-sm focus:outline-none focus:ring-2 focus:ring-indigo-500/20 focus:border-indigo-500"
                  />
                </div>

                <div className="flex justify-end">
                  <button
                    onClick={handleExplainTopic}
                    disabled={isExplaining || !topicName.trim()}
                    className="inline-flex items-center gap-2 bg-indigo-600 hover:bg-indigo-700 disabled:opacity-50 text-white font-medium px-5 py-2.5 rounded-xl text-sm transition-all shadow-sm shadow-indigo-200"
                  >
                    {isExplaining ? (
                      <>
                        <RefreshCw className="w-4 h-4 animate-spin" />
                        Anlatım Hazırlanıyor...
                      </>
                    ) : (
                      <>
                        <Feather className="w-4 h-4" />
                        Basitçe Anlat
                      </>
                    )}
                  </button>
                </div>
              </div>

              {/* Result Area */}
              {topicResult && (
                <div className="bg-white rounded-2xl p-6 border border-slate-200/80 shadow-sm space-y-3">
                  <div className="flex items-center justify-between border-b border-slate-100 pb-3">
                    <span className="text-xs font-semibold text-emerald-600 uppercase tracking-wider flex items-center gap-1.5">
                      <BookOpen className="w-4 h-4 text-emerald-500" /> Özel Konu Rehberi
                    </span>
                    <button
                      onClick={() => handleCopy(topicResult)}
                      className="text-xs text-slate-400 hover:text-slate-600 flex items-center gap-1 bg-slate-50 hover:bg-slate-100 px-2.5 py-1 rounded-lg transition-all"
                    >
                      {copied ? <Check className="w-3.5 h-3.5 text-emerald-500" /> : <Copy className="w-3.5 h-3.5" />}
                      {copied ? "Kopyalandı" : "Kopyala"}
                    </button>
                  </div>
                  <SimpleMarkdown content={topicResult} />
                </div>
              )}
            </div>
          )}

          {/* TAB 4: Metin Özetleme */}
          {activeTab === 'summary' && (
            <div className="space-y-6 animate-fadeIn">
              <header className="space-y-1">
                <div className="inline-flex items-center gap-1.5 px-2.5 py-1 rounded-full bg-amber-50 text-amber-600 text-xs font-semibold">
                  <FileText className="w-3.5 h-3.5" /> Hızlı Ders Notu
                </div>
                <h2 className="text-2xl font-bold text-slate-900">Metin Özetleme</h2>
                <p className="text-slate-500 text-sm">Uzun ders metinlerini veya makaleleri yapıştırın, anahtar noktaları hemen çıkaralım.</p>
              </header>

              <div className="bg-white rounded-2xl p-5 border border-slate-200/80 shadow-sm space-y-4">
                <div>
                  <label className="block text-xs font-semibold text-slate-600 mb-1.5">Özetleme Formatı</label>
                  <div className="grid grid-cols-3 gap-2">
                    <button
                      type="button"
                      onClick={() => setSummaryStyle('maddeler')}
                      className={`py-2 px-3 text-xs font-medium rounded-xl border transition-all ${
                        summaryStyle === 'maddeler'
                          ? 'bg-indigo-50 border-indigo-200 text-indigo-600'
                          : 'bg-slate-50 border-slate-200 text-slate-600 hover:bg-slate-100'
                      }`}
                    >
                      📌 Maddeler Halinde
                    </button>
                    <button
                      type="button"
                      onClick={() => setSummaryStyle('kisa')}
                      className={`py-2 px-3 text-xs font-medium rounded-xl border transition-all ${
                        summaryStyle === 'kisa'
                          ? 'bg-indigo-50 border-indigo-200 text-indigo-600'
                          : 'bg-slate-50 border-slate-200 text-slate-600 hover:bg-slate-100'
                      }`}
                    >
                      ⚡ Ultra Kısa Paragraf
                    </button>
                    <button
                      type="button"
                      onClick={() => setSummaryStyle('kavramlar')}
                      className={`py-2 px-3 text-xs font-medium rounded-xl border transition-all ${
                        summaryStyle === 'kavramlar'
                          ? 'bg-indigo-50 border-indigo-200 text-indigo-600'
                          : 'bg-slate-50 border-slate-200 text-slate-600 hover:bg-slate-100'
                      }`}
                    >
                      🔑 Kilit Kavramlar
                    </button>
                  </div>
                </div>

                <div>
                  <label className="block text-xs font-semibold text-slate-600 mb-1.5">Özetlenecek Metin</label>
                  <textarea
                    rows={6}
                    value={summaryText}
                    onChange={(e) => setSummaryText(e.target.value)}
                    placeholder="Ders kitabı sayfasını, kitap özetini veya uzun ders notlarını buraya yapıştırın..."
                    className="w-full bg-slate-50 border border-slate-200 rounded-xl p-3 text-sm focus:outline-none focus:ring-2 focus:ring-indigo-500/20 focus:border-indigo-500 transition-all resize-y"
                  ></textarea>
                </div>

                <div className="flex justify-end">
                  <button
                    onClick={handleSummarize}
                    disabled={isSummarizing || !summaryText.trim()}
                    className="inline-flex items-center gap-2 bg-indigo-600 hover:bg-indigo-700 disabled:opacity-50 text-white font-medium px-5 py-2.5 rounded-xl text-sm transition-all shadow-sm shadow-indigo-200"
                  >
                    {isSummarizing ? (
                      <>
                        <RefreshCw className="w-4 h-4 animate-spin" />
                        Özetleniyor...
                      </>
                    ) : (
                      <>
                        <FileText className="w-4 h-4" />
                        Metni Özetle
                      </>
                    )}
                  </button>
                </div>
              </div>

              {/* Result Area */}
              {summaryResult && (
                <div className="bg-white rounded-2xl p-6 border border-slate-200/80 shadow-sm space-y-3">
                  <div className="flex items-center justify-between border-b border-slate-100 pb-3">
                    <span className="text-xs font-semibold text-amber-600 uppercase tracking-wider flex items-center gap-1.5">
                      <FileText className="w-4 h-4 text-amber-500" /> Ders Özeti
                    </span>
                    <button
                      onClick={() => handleCopy(summaryResult)}
                      className="text-xs text-slate-400 hover:text-slate-600 flex items-center gap-1 bg-slate-50 hover:bg-slate-100 px-2.5 py-1 rounded-lg transition-all"
                    >
                      {copied ? <Check className="w-3.5 h-3.5 text-emerald-500" /> : <Copy className="w-3.5 h-3.5" />}
                      {copied ? "Kopyalandı" : "Kopyala"}
                    </button>
                  </div>
                  <SimpleMarkdown content={summaryResult} />
                </div>
              )}
            </div>
          )}

        </div>
      </main>
    </div>
  );
}
